file(GLOB PLUGINS CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} *)
foreach(PLUGIN ${PLUGINS})
    if(IS_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/${PLUGIN})
        message(STATUS "Found plugin: ${PLUGIN}")
        file(GLOB_RECURSE PLUGIN_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/${PLUGIN}/*.c)
        add_library(${PLUGIN} MODULE ${PLUGIN_SOURCES})
        target_link_libraries(${PLUGIN} PRIVATE synvixa)
        target_include_directories(${PLUGIN} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/${PLUGIN})
    endif()
endforeach()
